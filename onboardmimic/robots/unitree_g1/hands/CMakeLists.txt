cmake_minimum_required(VERSION 3.10)
project(onboardmimic_pico_hands)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
add_executable(hand_core_test core_test.cpp)
target_compile_options(hand_core_test PRIVATE -UNDEBUG)
enable_testing()
add_test(NAME pico_hand_admission COMMAND hand_core_test)
option(BUILD_DDS_HANDS "Build installed Unitree BrainCo DDS adapter" OFF)
if(BUILD_DDS_HANDS)
  set(UNITREE_SDK_ROOT "$ENV{HOME}/unitree_sdk2" CACHE PATH "Installed SDK2 source root")
  find_path(UNITREE_IDL_INCLUDE unitree/idl/go2/MotorCmds_.hpp HINTS "${UNITREE_SDK_ROOT}/include" REQUIRED)
  find_path(JSON_INCLUDE_DIR nlohmann/json.hpp HINTS "${CMAKE_SOURCE_DIR}/../../../../var/deps/include" REQUIRED)
  find_library(UNITREE_HAND_SDK2 unitree_sdk2 HINTS "${UNITREE_SDK_ROOT}/lib/${CMAKE_SYSTEM_PROCESSOR}" REQUIRED)
  find_library(DDSC ddsc HINTS "${UNITREE_SDK_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}" REQUIRED)
  find_library(DDSCXX ddscxx HINTS "${UNITREE_SDK_ROOT}/thirdparty/lib/${CMAKE_SYSTEM_PROCESSOR}" REQUIRED)
  add_executable(pico_hand_bridge bridge.cpp)
  target_include_directories(pico_hand_bridge PRIVATE ${JSON_INCLUDE_DIR} ${UNITREE_IDL_INCLUDE}
    "${UNITREE_SDK_ROOT}/thirdparty/include" "${UNITREE_SDK_ROOT}/thirdparty/include/ddscxx"
    /usr/local/include /usr/local/include/ddscxx)
  target_link_libraries(pico_hand_bridge ${UNITREE_HAND_SDK2} ${DDSCXX} ${DDSC} zmq pthread)
endif()
